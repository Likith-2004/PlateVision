# Linux deployment pins. Use this INSTEAD of requirements.txt on a host.
#
#     pip install -r requirements-deploy.txt
#
# Why it exists: on Linux, `pip install torch` from PyPI pulls the CUDA build,
# which is ~2 GB of GPU libraries this project never uses. That alone fails the
# build on most platforms. The index below serves CPU-only wheels (~200 MB),
# and the "+cpu" pin makes pip take them from there rather than PyPI.
#
# EasyOCR is omitted: PaddleOCR is the default engine and dropping the spare
# saves ~109 MB. Add `easyocr==1.7.2` back if you want OCR_ENGINE=easyocr
# available as a fallback.

--extra-index-url https://download.pytorch.org/whl/cpu

# OCR. The PyPI `paddlepaddle` package is already the CPU build
# (`paddlepaddle-gpu` is the CUDA one).
#
# paddlex MUST be pinned: paddleocr 3.1.0 only declares `paddlex>=3.1.0`, so an
# unpinned install pulls the newest paddlex (e.g. 3.7.2), whose changed
# PaddlePredictorOption API breaks paddleocr 3.1.0 at construction with
# "PaddlePredictorOption.__init__() takes 1 positional argument but 2 were
# given". 3.1.3 is the version this project is verified against.
paddleocr==3.1.0
paddlepaddle==3.1.0
paddlex==3.1.3

# paddlex 3.1.3 pulls in langchain (it declares only `langchain>=0.2`) and does
# `from langchain.docstore.document import Document`. langchain 1.x removed
# langchain.docstore, so an unpinned install (which grabs langchain 1.3.x) fails
# at import with "No module named 'langchain.docstore'". Pin the whole langchain
# family to the 0.3.x line this project is verified against; the versions below
# are mutually consistent (taken from the working environment).
langchain==0.3.27
langchain-core==0.3.72
langchain-community==0.3.27
langchain-text-splitters==0.3.9
langchain-openai==0.3.28
langsmith==0.4.8
